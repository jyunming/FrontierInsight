\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits egin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
\usepackage{hyperref}
\usepackage{xurl}
% URLs in the body font and breakable at ANY character via xurl, with
% links shown without the default colored boxes. With pandoc's
% autolink_bare_uris (generation/paper.py), long reference URLs wrap
% inside the text column instead of running into the right margin.
\urlstyle{same}
\hypersetup{hidelinks}
\setlength{\emergencystretch}{3em}
\usepackage{amsmath, amssymb}
\usepackage{booktabs}

% xcolor for the muted Frontier Insight footer wordmark (see brandfoot

\usepackage{xcolor}

% Pandoc-compatibility preamble. The standard pandoc latex template
% defines these commands and the Shaded env; when we override with a
% custom template, we must re-declare them ourselves or pdflatex
% errors on the first encounter.
%
% Variable names below are NOT spelled out in this comment because
% pandoc would substitute them in the comment itself, and multi-line
% expansion would break the comment boundary and tear down the
% preamble (see also templates/poster/poster.tex for the same trap).
%
% \tightlist is emitted inside markdown compact lists.
% \passthrough is emitted around inline raw-blocks.
% The highlighting-macros variable below expands to the Shaded /
% Highlighting envs + per-token color commands; expands to no-op
% when paper.md has no code blocks.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

% Default \maketitle on article class sets the title at \LARGE
% (~17.3pt at 11pt base), which dominates page 1 and looks oversized
% next to "Frontier Insight" below it. Drop to \Large (~14.4pt) so
% the title is still emphasized but no longer crowds the abstract
% block. titling lets us redefine the title spacing without
% re-implementing \maketitle.
\pretitle{\begin{center}\Large\bfseries}
\posttitle{\par\end{center}\vskip 0.5em}
\preauthor{\begin{center}\normalsize}
\postauthor{\par\end{center}}
\predate{\begin{center}\normalsize}
\postdate{\par\end{center}}

\title{$title$}
\author{Frontier Insight}
\date{$date$}

% PDF metadata: feed the pandoc title variable directly into
% hyperref so the rendered PDF carries a proper Title field
% (pdfinfo / Zotero / Mendeley all key on this — leaving it empty
% breaks reference-manager import). The variable name is NOT
% spelled out in this comment — see the matching note in the
% preamble at the top of this file. Pandoc would otherwise
% substitute it INSIDE the comment, the title's natural line
% wrap would break the comment boundary, and the leftover lines
% would land as raw LaTeX outside \begin{document}, killing the
% compile with "Missing \begin{document}".
\hypersetup{pdftitle={$title$}, pdfauthor={Frontier Insight}}

\renewcommand{\headrulewidth}{0pt}
$endif$
