% Pandoc-compatibility preamble. The standard pandoc latex template
% defines these commands and the Shaded env; when we override with a
% custom template, we must re-declare them ourselves or pdflatex
% errors on the first encounter.
%
% Variable names below are NOT spelled out in this comment because
% pandoc would substitute them in the comment itself, and multi-line
% expansion would break the comment boundary and tear down the
% preamble (see also templates/poster/poster.tex for the same trap).
%
% \tightlist is emitted inside markdown compact lists.
% \passthrough is emitted around inline raw-blocks.
% The highlighting-macros variable below expands to the Shaded /
% Highlighting envs + per-token color commands; expands to no-op
% when paper.md has no code blocks.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

% Default \maketitle on article class sets the title at \LARGE
% (~17.3pt at 11pt base), which dominates page 1 and looks oversized
% next to "Frontier Insight" below it. Drop to \Large (~14.4pt) so
% the title is still emphasized but no longer crowds the abstract
% block. titling lets us redefine the title spacing without
% re-implementing \maketitle.
\pretitle{\begin{center}\Large\bfseries}
\posttitle{\par\end{center}\vskip 0.5em}
\preauthor{\begin{center}\normalsize}
\postauthor{\par\end{center}}
\predate{\begin{center}\normalsize}
\postdate{\par\end{center}}

% PDF metadata: hyperref pulls pdftitle from \title{} only when
% \hypersetup is called AFTER \title — order matters. Without this
% block the rendered PDF has an empty Title field in pdfinfo, which
% breaks reference managers (Zotero, Mendeley) that key on it.
\hypersetup{pdftitle={$title$}, pdfauthor={Frontier Insight}}

\title{$title$}
\author{Frontier Insight}
\date{$date$}

\begin{document}
\maketitle

$if(abstract)$
\begin{abstract}
$abstract$
\end{abstract}
$endif$

$body$

\end{document}
