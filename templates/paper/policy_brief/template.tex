% FrontierInsight non-scientific paper template — policy brief.
% 2-4 page hard limit. The brief is the entire deliverable; there
% is no longer paper to back it up. Pandoc will respect h1 → section
% but the prompt should keep the brief to ~5 H1s max. No TOC. Tight
% margins so the brief reads dense without being cramped.

\documentclass[10pt,a4paper]{article}
\usepackage[utf8]{inputenc}
\usepackage[margin=0.85in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits egin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
\usepackage{amsmath, amssymb}
\usepackage{hyperref}
\usepackage{xurl}
% URLs in the body font and breakable at ANY character via xurl, with
% links shown without the default colored boxes. With pandoc's
% autolink_bare_uris (generation/paper.py), long reference URLs wrap
% inside the text column instead of running into the right margin.
\urlstyle{same}
\hypersetup{hidelinks}
\setlength{\emergencystretch}{3em}
\usepackage{titlesec}
\usepackage{booktabs}
\usepackage{xcolor}

% Pandoc-compatibility preamble — see generic/template.tex.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

% Bold section headings to anchor the brief's three-act shape:
% "Issue" → "Context" → "Recommendation". Pandoc maps Markdown h1
% to \section; configure those for visual weight.
\titleformat{\section}{\normalfont\large\bfseries\color{black!85}}{\thesection}{0.6em}{}
\titlespacing*{\section}{0pt}{1.0em}{0.4em}

% Header strip with the title + date so readers know what they're
% holding when the brief gets printed and detached from any email.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\lhead{\small\textbf{Policy Brief} — $title$}
\rhead{\small $date$}
\rfoot{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
% Subtle Frontier Insight brand mark, bottom-left foot of every page:
% the palette-native teal glyph + teal wordmark (matches the slide
% corner mark). Set by generation/paper.py only when the mark asset is
% present; no-ops otherwise.
$if(brandfoot)$
\lfoot{\raisebox{-0.30\height}{\includegraphics[height=12pt]{fi_icon.png}}\hspace{4pt}\textcolor[HTML]{0E6E6B}{\small Frontier Insight}}
$endif$

\title{\Large\bfseries $title$}
\author{}
\date{}

\begin{document}
{\Large\bfseries $title$ \par}
\vspace{0.4em}
{\small Frontier Insight — $date$ \par}
\vspace{1.2em}

$body$

\end{document}
