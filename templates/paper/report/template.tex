% FrontierInsight non-scientific paper template — executive report.
% Consulting-style: cover page with title, TOC after the cover,
% sans-serif body, page numbers in the footer. The Markdown body
% should open with an "Executive Summary" H1 — pandoc renders that
% as the first section after the TOC.

\documentclass[11pt,a4paper]{article}
\usepackage[utf8]{inputenc}
\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits egin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
\usepackage{amsmath, amssymb}
\usepackage{hyperref}
\usepackage{xurl}
% URLs in the body font and breakable at ANY character via xurl, with
% links shown without the default colored boxes. With pandoc's
% autolink_bare_uris (generation/paper.py), long reference URLs wrap
% inside the text column instead of running into the right margin.
\urlstyle{same}
\hypersetup{hidelinks}
\setlength{\emergencystretch}{3em}
% helvet is universally available in minimal TeX Live installs;
% opensans (an earlier choice) is not, and tripped paper_pdf compile
% on stripped containers.
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{booktabs}
% xcolor for the muted Frontier Insight footer wordmark (see brandfoot
% block below). hyperref alone does not expose \textcolor.
\usepackage{xcolor}

% Pandoc-compatibility preamble — see generic/template.tex.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

\pagestyle{fancy}
\fancyhf{}
\rfoot{\thepage}
\renewcommand{\headrulewidth}{0pt}
% Subtle Frontier Insight brand mark, bottom-left foot of every page:
% the palette-native teal glyph + teal wordmark (matches the slide
% corner mark). Set by generation/paper.py only when the mark asset was
% copied next to the source; no-ops otherwise.
$if(brandfoot)$
\lfoot{\raisebox{-0.30\height}{\includegraphics[height=12pt]{fi_icon.png}}\hspace{4pt}\textcolor[HTML]{0E6E6B}{\small Frontier Insight}}
$endif$

\title{\Huge\bfseries $title$}
\author{Frontier Insight}
\date{$date$}

\begin{document}
\begin{titlepage}
  \maketitle
  \thispagestyle{empty}
\end{titlepage}

\tableofcontents
\clearpage

$body$

\end{document}
