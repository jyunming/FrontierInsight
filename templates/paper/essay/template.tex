% FrontierInsight non-scientific paper template — essay format.
% Long-form argumentative prose for cultural / historical /
% qualitative topics where IMRAD doesn't fit. Wider margins, serif
% body, larger leading. No abstract block — the essay should land
% its thesis in the opening paragraph instead.

\documentclass[12pt,a4paper]{article}
\usepackage[utf8]{inputenc}
\usepackage[margin=1.4in,top=1.2in,bottom=1.2in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits egin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
% Required for pandoc-rendered markdown tables. Pandoc emits \toprule /
% \midrule / \bottomrule around every pipe table; without booktabs the
% compile dies on ``Undefined control sequence. \toprule`` at the first
% table.
\usepackage{booktabs}
\usepackage{amsmath, amssymb}
\usepackage{hyperref}
\usepackage{setspace}
\usepackage{microtype}

% Pandoc-compatibility preamble — see generic/template.tex.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

\onehalfspacing

\title{$title$}
\author{Frontier Insight}
\date{$date$}

\begin{document}
\maketitle

$body$

\end{document}
