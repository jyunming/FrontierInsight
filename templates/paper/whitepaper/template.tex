% FrontierInsight non-scientific paper template — whitepaper.
% Industry / vendor-neutral analysis: problem → approach → evidence
% → conclusions. Cover page with subtitle slot, sans-serif body,
% modest TOC for the medium-length doc (8-20 pages typical). Avoids
% the academic "Authors / Affiliations" line — whitepapers are
% organizational outputs, not bylined.

\documentclass[11pt,a4paper]{article}
% pdflatex cannot set Chinese, Japanese or Korean text; generation/paper.py
% compiles such a paper with XeLaTeX and passes fi-cjk-font.
\usepackage{iftex}
\ifXeTeX
  \usepackage{fontspec}
$if(fi-cjk-font)$
  \usepackage{xeCJK}
  \setCJKmainfont{$fi-cjk-font$}
  \setCJKsansfont{$fi-cjk-font$}
  \setCJKmonofont{$fi-cjk-font$}
$endif$
\else
  \usepackage[utf8]{inputenc}
\fi
\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits \begin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
\usepackage{amsmath, amssymb}
\usepackage{hyperref}
% Break long URLs (e.g. reference links) at ANY character so they wrap
% inside the text column instead of running into the right margin — the
% common "reference URL overflows the page" failure. xurl must load
% after hyperref. emergencystretch gives TeX extra slack to avoid other
% overfull lines (long unbreakable tokens, wide inline code).
\usepackage{xurl}
% URLs in the body font (not monospace) and breakable at ANY character
% via xurl, with links shown without the default colored boxes. Combined
% with pandoc's autolink_bare_uris (set in generation/paper.py), this
% wraps long reference URLs inside the text column instead of running
% them into the right margin (the "reference URL overflow" failure).
\urlstyle{same}
\hypersetup{hidelinks}
\setlength{\emergencystretch}{3em}
% helvet is universally available in minimal TeX Live installs;
% opensans (an earlier choice) is not, and tripped paper_pdf compile
% on stripped containers. Under XeLaTeX (a paper with CJK text) the Type 1
% helvet would fall back to Latin Modern, so the OpenType Helvetica clone
% TeX Gyre Heros stands in.
\ifXeTeX
  \setsansfont{TeX Gyre Heros}
\else
  \usepackage{helvet}
\fi
\renewcommand{\familydefault}{\sfdefault}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{booktabs}
\usepackage{xcolor}

% Pandoc-compatibility preamble — see generic/template.tex.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
% A figure fills the line width (its column, in two columns) but is never
% taller than 45% of the text height, so it shares a page with text rather
% than being moved on to the next page and leaving this one half empty.
\makeatletter
\newsavebox{\FI@figbox}
\providecommand{\pandocbounded}[1]{%
  \sbox{\FI@figbox}{#1}%
  \Gscale@div\@tempa{0.45\textheight}{\dimexpr\ht\FI@figbox+\dp\FI@figbox\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\FI@figbox}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \noindent\scalebox{\@tempa}{\usebox{\FI@figbox}}}
% A figure may also stay where it is written (h), and a page of floats must
% be 80% full, so a figure is never put alone on a page half of it blank.
\def\fps@figure{htbp}
\makeatother
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.6}
\renewcommand{\textfraction}{0.1}
\renewcommand{\floatpagefraction}{0.8}
$highlighting-macros$

% Slightly heavier section style than `report` — whitepapers benefit
% from clearer hierarchy when the doc spans 10+ pages.
\titleformat{\section}{\Large\bfseries\color{black!90}}{\thesection}{0.6em}{}
\titleformat{\subsection}{\large\bfseries\color{black!75}}{\thesubsection}{0.5em}{}

\pagestyle{fancy}
\fancyhf{}
\lhead{\small\textit{$title$}}
\rfoot{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
% Subtle Frontier Insight brand mark, bottom-left foot of every page:
% the palette-native teal glyph + teal wordmark (matches the slide
% corner mark). `brandfoot` is set by generation/paper.py only when the
% mark asset was copied next to the source, so this no-ops cleanly on
% hosts without the asset.
$if(brandfoot)$
\lfoot{\raisebox{-0.30\height}{\includegraphics[height=12pt]{fi_icon.png}}\hspace{4pt}\textcolor[HTML]{0E6E6B}{\small Frontier Insight}}
$endif$

\title{\Huge\bfseries $title$}
% Author block from the output.author fields (see generic/template.tex).
\author{$if(fi-author)$$fi-author$$else$Frontier Insight$endif$$if(fi-affiliation)$\\{\small $fi-affiliation$}$endif$$if(fi-email)$\\{\small $fi-email$}$endif$$if(fi-url)$\\{\small $fi-url$}$endif$}
\date{$date$}

% A visual check that finds a last page holding only a line or two
% recompiles with fi-extra-lines set: the text area grows that many lines and
% the footer moves up by the same amount, so it stays where it was
% (generation/_visual_check.py).
$if(fi-extra-lines)$
\AtBeginDocument{\addtolength{\textheight}{$fi-extra-lines$\baselineskip}\addtolength{\footskip}{-$fi-extra-lines$\baselineskip}}
$endif$

\begin{document}
\begin{titlepage}
  \vspace*{2cm}
  {\Huge\bfseries $title$ \par}
  \vspace{1cm}
  {\Large A Frontier Insight whitepaper \par}
$if(fi-author)$
  \vspace{0.8cm}
  {\large $fi-author$ \par}
$endif$
$if(fi-affiliation)$
  {\normalsize $fi-affiliation$ \par}
$endif$
$if(fi-email)$
  {\normalsize $fi-email$ \par}
$endif$
$if(fi-url)$
  {\normalsize $fi-url$ \par}
$endif$
  \vfill
  {\large $date$ \par}
\end{titlepage}

\tableofcontents
\clearpage

$body$

\end{document}
