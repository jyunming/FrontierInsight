% FrontierInsight non-scientific paper template — whitepaper.
% Industry / vendor-neutral analysis: problem → approach → evidence
% → conclusions. Cover page with subtitle slot, sans-serif body,
% modest TOC for the medium-length doc (8-20 pages typical). Avoids
% the academic "Authors / Affiliations" line — whitepapers are
% organizational outputs, not bylined.

\documentclass[11pt,a4paper]{article}
\usepackage[utf8]{inputenc}
\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits egin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
\usepackage{amsmath, amssymb}
\usepackage{hyperref}
% Break long URLs (e.g. reference links) at ANY character so they wrap
% inside the text column instead of running into the right margin — the
% common "reference URL overflows the page" failure. xurl must load
% after hyperref. emergencystretch gives TeX extra slack to avoid other
% overfull lines (long unbreakable tokens, wide inline code).
\usepackage{xurl}
% URLs in the body font (not monospace) and breakable at ANY character
% via xurl, with links shown without the default colored boxes. Combined
% with pandoc's autolink_bare_uris (set in generation/paper.py), this
% wraps long reference URLs inside the text column instead of running
% them into the right margin (the "reference URL overflow" failure).
\urlstyle{same}
\hypersetup{hidelinks}
\setlength{\emergencystretch}{3em}
% helvet is universally available in minimal TeX Live installs;
% opensans (an earlier choice) is not, and tripped paper_pdf compile
% on stripped containers.
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{booktabs}
\usepackage{xcolor}

% Pandoc-compatibility preamble — see generic/template.tex.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

% Slightly heavier section style than `report` — whitepapers benefit
% from clearer hierarchy when the doc spans 10+ pages.
\titleformat{\section}{\Large\bfseries\color{black!90}}{\thesection}{0.6em}{}
\titleformat{\subsection}{\large\bfseries\color{black!75}}{\thesubsection}{0.5em}{}

\pagestyle{fancy}
\fancyhf{}
\lhead{\small\textit{$title$}}
\rfoot{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
% Subtle Frontier Insight brand mark, bottom-left foot of every page:
% the palette-native teal glyph + teal wordmark (matches the slide
% corner mark). `brandfoot` is set by generation/paper.py only when the
% mark asset was copied next to the source, so this no-ops cleanly on
% hosts without the asset.
$if(brandfoot)$
\lfoot{\raisebox{-0.30\height}{\includegraphics[height=12pt]{fi_icon.png}}\hspace{4pt}\textcolor[HTML]{0E6E6B}{\small Frontier Insight}}
$endif$

\title{\Huge\bfseries $title$}
\author{Frontier Insight}
\date{$date$}

\begin{document}
\begin{titlepage}
  \vspace*{2cm}
  {\Huge\bfseries $title$ \par}
  \vspace{1cm}
  {\Large A Frontier Insight whitepaper \par}
  \vfill
  {\large $date$ \par}
\end{titlepage}

\tableofcontents
\clearpage

$body$

\end{document}
