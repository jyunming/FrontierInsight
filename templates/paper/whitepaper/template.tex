% FrontierInsight non-scientific paper template — whitepaper.
% Industry / vendor-neutral analysis: problem → approach → evidence
% → conclusions. Cover page with subtitle slot, sans-serif body,
% modest TOC for the medium-length doc (8-20 pages typical). Avoids
% the academic "Authors / Affiliations" line — whitepapers are
% organizational outputs, not bylined.

\documentclass[11pt,a4paper]{article}
\usepackage[utf8]{inputenc}
\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
% Required for pandoc-rendered markdown tables (pandoc emits egin{longtable}).
\usepackage{longtable}
% Required for pandoc-rendered table column widths. Pandoc emits column
% specs like ``p{(\linewidth - 6\tabcolsep) * \real{0.2667}}`` whose
% arithmetic only parses with calc loaded. Without this, any markdown
% table with relative column widths fails the LaTeX compile with
% ``Missing number, treated as zero`` at the longtable \begin{}.
\usepackage{calc}
\usepackage{array}
% pandoc emits \def\LTcaptype{none} for caption-less tables; longtable
% then calls \refstepcounter{none} internally, so the counter must exist.
\newcounter{none}
\usepackage{amsmath, amssymb}
\usepackage{hyperref}
% helvet is universally available in minimal TeX Live installs;
% opensans (an earlier choice) is not, and tripped paper_pdf compile
% on stripped containers.
\usepackage{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{booktabs}
\usepackage{xcolor}

% Pandoc-compatibility preamble — see generic/template.tex.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

% Slightly heavier section style than `report` — whitepapers benefit
% from clearer hierarchy when the doc spans 10+ pages.
\titleformat{\section}{\Large\bfseries\color{black!90}}{\thesection}{0.6em}{}
\titleformat{\subsection}{\large\bfseries\color{black!75}}{\thesubsection}{0.5em}{}

\pagestyle{fancy}
\fancyhf{}
\lhead{\small\textit{$title$}}
\rfoot{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}

\title{\Huge\bfseries $title$}
\author{Frontier Insight}
\date{$date$}

\begin{document}
\begin{titlepage}
  \vspace*{2cm}
  {\Huge\bfseries $title$ \par}
  \vspace{1cm}
  {\Large A Frontier Insight whitepaper \par}
  \vfill
  {\large $date$ \par}
\end{titlepage}

\tableofcontents
\clearpage

$body$

\end{document}
