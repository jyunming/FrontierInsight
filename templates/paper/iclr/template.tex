% Minimal ICLR-flavored stand-in. Replace with the official
% iclr2024_conference.sty when you actually submit; this is enough
% for pandoc to compile paper.md with ICLR-style metadata blocks
% (single-column 10pt, abstract env, named author block).
\documentclass[10pt,letterpaper]{article}
\usepackage[utf8]{inputenc}
\usepackage[margin=1in]{geometry}
\usepackage{graphicx}
\usepackage{hyperref}
\usepackage{amsmath, amssymb}
\usepackage{booktabs}
\usepackage{microtype}
% authblk + titling both patch \author/\maketitle and the combo trips
% LaTeX's "Missing \begin{document}" on \author[N]{}. Use titling alone.
\usepackage{titling}
\usepackage{titlesec}

% Pandoc-compatibility preamble — see generic/template.tex for the
% full rationale.
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\passthrough}[1]{#1}
\providecommand{\pandocbounded}[1]{\noindent\resizebox{\textwidth}{!}{#1}}
$highlighting-macros$

% ICLR's title block is centered and noticeably tighter than the
% article default; matching that visual rhythm here.
\pretitle{\begin{center}\Large\bfseries}
\posttitle{\par\end{center}\vskip 0.4em}
\preauthor{\begin{center}\normalsize}
\postauthor{\par\end{center}}

% ICLR uses bold section heads, no leading "Section" word.
\titleformat{\section}{\normalfont\large\bfseries}{\thesection}{0.6em}{}

\hypersetup{pdftitle={$title$}, pdfauthor={Frontier Insight}}

\title{$title$}
\author{Frontier Insight \\ \small\itshape Generated by an automated research pipeline}
\date{$date$}

\begin{document}
\maketitle

$if(abstract)$
\begin{abstract}
$abstract$
\end{abstract}
$endif$

$body$

\end{document}
