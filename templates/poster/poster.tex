% Frontier Insight — A1 landscape (84.1 x 59.4 cm = 33" x 23") 3-column
% poster template.
%
% Why A1 instead of 36"x48":
% The LLM-generated content (3 columns, ~3-4 figures) reliably fills
% A1 from edge to edge. The old 121.92 x 91.44 cm (48"x36") page left
% the bottom ~50% blank for short / medium-depth quests, which made
% posters look incomplete. A1 is a real, printable poster size
% (standard for single-author academic conference posters) and gives
% a dense, professional rendering for the content volume FI actually
% generates. Users who need a larger conference-hall poster can edit
% this file: set width=121.92 height=91.44 for 48"x36", or
% width=118.9 height=84.1 for A0.
%
% Substitutions (Python string.Template): title, left, middle, right.
% In the body below, each placeholder appears as a single-dollar
% ``$$name`` form — that is what safe_substitute() actually replaces.
% (The double-dollar ``$$$$name`` form you see in THIS comment is the
% string.Template escape: safe_substitute reduces ``$$$$`` to a literal
% ``$$`` and does NOT perform any replacement on it. Every dollar-
% prefixed variable name in the rest of THIS comment is therefore
% doubled on purpose so the comment renders verbatim instead of being
% substituted itself.)
% If you let a bare single-dollar ``$$name`` appear in this comment,
% multi-line LLM-generated content (which can contain its own
% \begin/end{column} markers) gets injected INTO this comment line;
% the second and later lines are no longer inside a "%" comment and
% end up as raw LaTeX BEFORE \documentclass — pdflatex then errors
% on \textbf with "Undefined control sequence".
\documentclass[final,t]{beamer}
\usepackage[size=custom,width=84.1,height=59.4,scale=1.0]{beamerposter}
\usepackage{graphicx}
\usepackage{multicol}
\usepackage{amsmath, amssymb}
\usepackage{lmodern}
\usepackage{hyperref}

\title{$title}
\author{Frontier Insight}
\institute{Generated by an automated research pipeline}

% Headline block — pinned to the top of the poster so the title +
% author + institute actually render. Without this, beamerposter's
% default headline is empty: \title{} is set but never displayed,
% which is what produced the headerless / half-blank posters users
% reported. The block spans the full paper width and uses Huge title
% font so the poster reads from across a room.
\setbeamertemplate{headline}{%
  \leavevmode%
  \begin{beamercolorbox}[wd=\paperwidth, sep=0.5em]{headline}%
    \begin{minipage}{\paperwidth}%
      \centering%
      \vspace{0.4em}%
      {\usebeamerfont{title}\Huge\bfseries\inserttitle\par}%
      \vspace{0.4em}%
      {\large\insertauthor\quad{\itshape\insertinstitute}\par}%
      \vspace{0.4em}%
    \end{minipage}%
  \end{beamercolorbox}%
  \vspace{0.4em}%
}

\begin{document}
\begin{frame}[t]
  \begin{columns}[t]
    \begin{column}{0.32\linewidth}
      $left
    \end{column}
    \begin{column}{0.32\linewidth}
      $middle
    \end{column}
    \begin{column}{0.32\linewidth}
      $right
    \end{column}
  \end{columns}

  % Sources band — injected by generation/poster.py from the quest's
  % retrieved literature (web pages + papers), NOT the LLM, so every
  % poster carries its citations even when the columns run long. Empty
  % string when the quest surfaced no citable sources.
  $references
\end{frame}
\end{document}
