% Frontier Insight — A1 PORTRAIT (59.4 x 84.1 cm = 23" x 33") 2-column
% poster template.
%
% Why portrait + 2 columns: academic conference posters are most often
% portrait, and a 2-column portrait layout gives each figure a wide,
% tall area — so the poster reads figure-first (big charts, short text)
% rather than as walls of prose. The web-derived plots are the focus;
% the LLM columns wrap each with a one-line takeaway. Users who want a
% landscape or larger size can edit the geometry below
% (A0 portrait: width=84.1 height=118.9; A1 landscape: width=84.1
% height=59.4 with three 0.32\linewidth columns).
%
% Substitutions (Python string.Template): title, left, right, references.
% In the body below each placeholder appears as a single-dollar ``$name``
% form — that is what safe_substitute() replaces. (The doubled ``$$name``
% form in THIS comment is the string.Template escape so the comment is
% not itself substituted; if a bare ``$name`` appeared here, multi-line
% LLM content could be injected into the comment and leak raw LaTeX
% before \documentclass.)
\documentclass[final,t]{beamer}
\usepackage[size=custom,width=59.4,height=84.1,scale=1.15]{beamerposter}
\usepackage{graphicx}
\usepackage{multicol}
\usepackage{amsmath, amssymb}
\usepackage{lmodern}
\usepackage{hyperref}

\title{$title}
\author{Frontier Insight}
\institute{Generated by an automated research pipeline}

% Headline block — pinned to the top so the title + author + institute
% render across the full width.
\setbeamertemplate{headline}{%
  \leavevmode%
  \begin{beamercolorbox}[wd=\paperwidth, sep=0.5em]{headline}%
    \begin{minipage}{\paperwidth}%
      \centering%
      \vspace{0.4em}%
      {\usebeamerfont{title}\Huge\bfseries\inserttitle\par}%
      \vspace{0.3em}%
      {\large\insertauthor\quad{\itshape\insertinstitute}\par}%
      \vspace{0.3em}%
    \end{minipage}%
  \end{beamercolorbox}%
  \vspace{0.4em}%
}

\begin{document}
\begin{frame}[t]
  \begin{columns}[t]
    \begin{column}{0.48\linewidth}
      $left
    \end{column}
    \begin{column}{0.48\linewidth}
      $right
    \end{column}
  \end{columns}

  % Sources band — injected by generation/poster.py from the quest's
  % retrieved literature (web pages + papers), NOT the LLM, so every
  % poster carries its citations even when the columns run long. Empty
  % string when the quest surfaced no citable sources.
  $references
\end{frame}
\end{document}
