% Frontier Insight research poster, filled in by generation/poster.py.
%
% The generator writes everything that carries content (the header text,
% the columns, the reference band) and takes the sheet size and the font
% sizes from output.poster_size. This file holds the page setup, fonts,
% palette and layout.
%
% Layout, following published poster guidance:
%   - header: brand line, the main finding as the headline, the paper's
%     own title under it, the author line, a QR code on the right when a
%     URL is set, closed by a full-width teal rule;
%   - columns of fixed height filled from the top: body text at 24 pt or
%     more, headings about 1.5 times the body, numbered figure captions;
%   - a footer band listing only the references the poster cites.
%
% Placeholders are string.Template names in braces after a dollar sign;
% a literal dollar sign in this file would have to be doubled.
\documentclass[final,t]{beamer}
\usepackage[size=custom,width=${paperwidth},height=${paperheight},scale=1.0]{beamerposter}
\usepackage{iftex}
\ifXeTeX
  % XeLaTeX: the poster holds Chinese, Japanese or Korean text, or the
  % engine is tectonic. OpenType clones of Palatino and Helvetica.
  \usepackage{fontspec}
  \setmainfont{TeX Gyre Pagella}
  \setsansfont{TeX Gyre Heros}
${cjksetup}
\else
  \usepackage[utf8]{inputenc}
  \usepackage[T1]{fontenc}
  \usepackage{newpxtext}
  \usepackage{helvet}
\fi
\usepackage{graphicx}
\usepackage{amsmath, amssymb}
\usepackage{xcolor}
\usepackage{multicol}
\usepackage{qrcode}
\usepackage{hyperref}

% --- Frontier Insight palette (matches templates/slides/fi.css) -------
\definecolor{fiteal}{HTML}{0E6E6B}
\definecolor{ficream}{HTML}{F8F7F3}
\definecolor{fiink}{HTML}{16222B}
\definecolor{fihair}{HTML}{C9C6BA}
\definecolor{fimuted}{HTML}{56636C}

\setbeamercolor{background canvas}{bg=ficream}
\setbeamercolor{normal text}{fg=fiink}
\setbeamercolor{headline}{bg=ficream,fg=fiink}
\setbeamercolor{footline}{bg=ficream,fg=fiink}
\usebeamercolor[fg]{normal text}
\renewcommand{\familydefault}{\sfdefault}
\setbeamertemplate{navigation symbols}{}
\setbeamersize{text margin left=${margin}cm, text margin right=${margin}cm}

% Type sizes come from the sheet: never below 24 pt body, 36 pt headings,
% 72 pt title, 18 pt captions and 14 pt references.
\newcommand{\posterbody}{\fontsize{${bodypt}pt}{${bodylead}pt}\selectfont}
\newcommand{\posterhead}[1]{%
  \par\vspace{0.4em}%
  {\fontsize{${headpt}pt}{${headlead}pt}\selectfont\bfseries\color{fiink}#1\par}%
  \vspace{0.2em}{\color{fiteal}\rule{0.14\linewidth}{0.16em}}\par\vspace{0.5em}}
\newcommand{\postercaption}[2]{%
  {\fontsize{${captionpt}pt}{${captionlead}pt}\selectfont\color{fimuted}\textbf{Figure~#1:}~#2\par}}
\newenvironment{posterlist}
  {\begin{list}{\color{fiteal}\rule[0.2em]{0.42em}{0.42em}}%
    {\setlength{\leftmargin}{1.3em}\setlength{\labelwidth}{0.8em}\setlength{\labelsep}{0.5em}%
     \setlength{\itemsep}{0.3em}\setlength{\topsep}{0pt}\setlength{\parsep}{0pt}%
     \setlength{\partopsep}{0pt}}}
  {\end{list}}

\setbeamertemplate{headline}{%
  \leavevmode
  \begin{beamercolorbox}[wd=\paperwidth]{headline}%
    \vskip${margin}cm
    \hspace*{${margin}cm}%
    \begin{minipage}[t]{${headerwidth}cm}%
      \raggedright
      ${brandline}%
      {\rmfamily\bfseries\fontsize{${titlept}pt}{${titlelead}pt}\selectfont\color{fiink}${headline}\par}%
      ${subtitle}%
      ${authorline}%
    \end{minipage}%
    ${qrcode}%
    \par\vskip0.9cm
    \hspace*{${margin}cm}{\color{fiteal}\rule{${rulewidth}cm}{0.3cm}}\par
    \vskip1.2cm
  \end{beamercolorbox}}

\setbeamertemplate{footline}{%
  \leavevmode
  \begin{beamercolorbox}[wd=\paperwidth]{footline}%
    \hspace*{${margin}cm}{\color{fihair}\rule{${rulewidth}cm}{2pt}}\par
    \vskip0.6cm
    \hspace*{${margin}cm}\begin{minipage}[t]{${rulewidth}cm}%
      \fontsize{${refspt}pt}{${refslead}pt}\selectfont\color{fiink}%
      ${references}%
    \end{minipage}\par
    \vskip${margin}cm
  \end{beamercolorbox}}

\begin{document}
\begin{frame}[t]\relax
${columns}
\end{frame}
\end{document}
