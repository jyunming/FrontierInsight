% Frontier Insight — A1 PORTRAIT (59.4 x 84.1 cm) premium editorial
% research poster.
%
% Design language ("Research Briefing", shared with the slide theme):
%   - warm cream page (#f8f7f3), ink text (#16222b), ONE accent = deep
%     teal (#0e6e6b); structural weight carried by hairline + teal rules,
%     never shadows or boxes.
%   - Palatino-class serif (newpx) for the display title; a clean sans
%     for the body; the Frontier Insight brand LOCKUP (teal glyph +
%     wordmark) pinned top-left of the headline band, with a thin teal
%     rule under the title.
%   - figure-first 2-column body; a full-width Sources band at the foot
%     carrying a second, smaller copy of the brand mark.
%
% Substitutions (Python string.Template): title, left, right,
% references, brandlogo. In the body each placeholder appears as a
% single-dollar ``$name`` form — that is what safe_substitute() replaces.
% (The doubled ``$$name`` form in THIS comment is the string.Template
% escape so the comment is not itself substituted; a bare ``$name`` here
% could let multi-line LLM content inject raw LaTeX before
% \documentclass.)
\documentclass[final,t]{beamer}
\usepackage[size=custom,width=59.4,height=84.1,scale=1.15]{beamerposter}
\usepackage{graphicx}
\usepackage{multicol}
\usepackage{amsmath, amssymb}
\usepackage{newpxtext}        % Palatino-class serif for the display title
\usepackage{helvet}           % Helvetica-class sans for the body
\usepackage{xcolor}
\usepackage{hyperref}

% --- Frontier Insight palette (matches templates/slides/fi.css) -------
\definecolor{fiteal}{HTML}{0E6E6B}
\definecolor{fitealdark}{HTML}{0A4F4D}
\definecolor{ficream}{HTML}{F8F7F3}
\definecolor{fiink}{HTML}{16222B}
\definecolor{fihair}{HTML}{D9D7CE}
\definecolor{fimuted}{HTML}{65727C}

% Cream page, ink text, teal list markers.
\setbeamercolor{background canvas}{bg=ficream}
\setbeamercolor{normal text}{fg=fiink}
\setbeamercolor{headline}{bg=ficream,fg=fiink}
\setbeamercolor{itemize item}{fg=fiteal}
\setbeamercolor{itemize subitem}{fg=fiteal}
\usebeamercolor[fg]{normal text}

% Sans body (matches the design system's sans body); the serif Palatino
% is reserved for the display title via \rmfamily in the headline below.
\renewcommand{\familydefault}{\sfdefault}

% Sharp teal square list markers — editorial, no rounded shapes.
\setbeamertemplate{itemize item}{\color{fiteal}\rule[0.12ex]{0.55ex}{0.55ex}}
\setbeamertemplate{itemize subitem}{\color{fitealdark}\textendash}
% Drop the tiny navigation symbols — they serve no purpose on a printed
% poster and could confuse the bottom-margin overflow detector.
\setbeamertemplate{navigation symbols}{}

\title{$title}
\author{Frontier Insight}

% Headline band: brand lockup + tag, hairline rule, big serif title,
% heavy teal rule. Pinned to the top across the full width.
\setbeamertemplate{headline}{%
  \leavevmode%
  \begin{beamercolorbox}[wd=\paperwidth]{headline}%
    \vskip0.5em%
    \begin{center}%
    \begin{minipage}{0.93\paperwidth}%
      \mbox{$brandlogo}\hfill{\color{fimuted}\large Automated research briefing}\\[0.45em]
      {\color{fihair}\rule{\linewidth}{1.2pt}}\\[0.6em]
      {\rmfamily\bfseries\color{fiink}\Huge \inserttitle\par}
      \vskip0.4em
      {\color{fiteal}\rule{\linewidth}{6pt}}
    \end{minipage}%
    \end{center}%
    \vskip0.4em%
  \end{beamercolorbox}%
}

\begin{document}
\begin{frame}[t]
  \begin{columns}[t]
    \begin{column}{0.47\linewidth}
      $left
    \end{column}
    \begin{column}{0.47\linewidth}
      $right
    \end{column}
  \end{columns}

  % Sources band — injected by generation/poster.py from the quest's
  % retrieved literature (web pages + papers), NOT the LLM, so every
  % poster carries its citations even when the columns run long. The
  % generator prepends a small teal Frontier Insight glyph to the start
  % of the band (a second, quiet copy of the header lockup). Empty
  % string when the quest surfaced no citable sources.
  $references
\end{frame}
\end{document}
